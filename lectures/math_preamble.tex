% Inline micro-exercise, numbered per section. Optional argument tags it, e.g.
% \begin{exercise}[optional stretch] for the skippable harder tier.
\newcounter{exercise}[section]
\renewcommand{\theexercise}{\thesection.\arabic{exercise}}
\newenvironment{exercise}[1][]{%
    \refstepcounter{exercise}%
    \par\smallskip\noindent\textbf{Exercise~\theexercise}%
    \if\relax\detokenize{#1}\relax\else\ \textit{(#1)}\fi
    \textbf{.}\enspace\ignorespaces
}{\par\smallskip}
